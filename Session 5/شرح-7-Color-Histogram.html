١ ما هو هيستوغرام الصورة؟

الهيستوغرام (Histogram) هو رسم بياني يُظهر توزيع قيم البيكسلات في الصورة. يُجيب على سؤال: "كم عدد البيكسلات التي قيمتها بين X وY؟"

مثال: صورة ليلية ستكون هيستوغرامها مُتمركزاً في اليسار (قيم داكنة قريبة من 0)، بينما صورة في ضوء الشمس ستكون هيستوغرامها موزعاً وممتداً نحو اليمين (قيم فاتحة قريبة من 255).

عندما نحسب الهيستوغرام لكل قناة لون على حدة (أحمر، أخضر، أزرق)، نحصل على وصف غني للتوزيع اللوني في الصورة — وهذا هو هيستوغرام الألوان.

٢ نظام الألوان BGR في OpenCV

🔵
Blue (B)
القناة الزرقاء — الفهرس 0
🟢
Green (G)
القناة الخضراء — الفهرس 1
🔴
Red (R)
القناة الحمراء — الفهرس 2
⚠️ تنبيه مهم: OpenCV يقرأ الصور بترتيب BGR وليس RGB! الصور الملونة في NumPy لها أبعاد (H, W, 3) حيث:
— الفهرس 0 = الأزرق
— الفهرس 1 = الأخضر
— الفهرس 2 = الأحمر
لو استخدمنا plt.imshow مباشرةً على صورة OpenCV ستظهر الألوان خاطئة (أحمر وأزرق معكوسان).

٣ شرح الكود الأول — رسم الهيستوغرام

import cv2
import matplotlib.pyplot as plt

# قراءة الصورة
img = cv2.imread('person.jpg')

# فصل القنوات
colors = ('b', 'g', 'r')

plt.figure()
plt.title("Color Histogram")
plt.xlabel("Pixel Intensity")
plt.ylabel("Count")

# حساب الهيستوغرام لكل قناة
for i, color in enumerate(colors):
    hist = cv2.calcHist([img], [i], None, [256], [0, 256])
    plt.plot(hist, color=color)

plt.show()

🔹 colors = ('b', 'g', 'r')

قائمة ألوان للرسم. الترتيب يطابق ترتيب BGR في OpenCV: 'b' للأزرق (الفهرس 0)، 'g' للأخضر (1)، 'r' للأحمر (2).

🔹 enumerate(colors)

تُرجع كلاً من الفهرس والقيمة معاً في كل تكرار: i=0, color='b' ثم i=1, color='g' ثم i=2, color='r'

🔹 cv2.calcHist([img], [i], None, [256], [0, 256])

المعامل الأول [img]: الصور — قائمة (حتى لو صورة واحدة)
المعامل الثاني [i]: فهرس القناة — [0] للأزرق، [1] للأخضر، [2] للأحمر
المعامل الثالث None: القناع — None يعني حساب الهيستوغرام على الصورة كاملة
المعامل الرابع [256]: عدد الخانات — 256 خانة (قيمة لكل بيكسل ممكنة)
المعامل الخامس [0, 256]: نطاق القيم — من 0 إلى 255

ناتج calcHist:

مصفوفة شكلها (256, 1) — كل عنصر هو عدد البيكسلات بهذه القيمة في هذه القناة.

٤ شرح الكود الثاني — إنشاء متجه الميزات

import numpy as np

# دمج القنوات الثلاث
hist_b = cv2.calcHist([img], [0], None, [256], [0,256])
hist_g = cv2.calcHist([img], [1], None, [256], [0,256])
hist_r = cv2.calcHist([img], [2], None, [256], [0,256])

# دمجهم في متجه واحد
feature_vector = np.concatenate([hist_b, hist_g, hist_r])

print(feature_vector.shape)  # (768, 1)
(768, 1)

🔹 حساب الهيستوغرام لكل قناة على حدة

نحسب هيستوغراماً منفصلاً لكل قناة:

المتغيرالقناةالفهرسالحجم
hist_bأزرق (Blue)[0](256, 1)
hist_gأخضر (Green)[1](256, 1)
hist_rأحمر (Red)[2](256, 1)

🔹 np.concatenate([hist_b, hist_g, hist_r])

دمج الثلاثة هيستوغرامات في متجه واحد طويل: 256 + 256 + 256 = 768 قيمة.

✅ متجه الميزات النهائي بطول 768 يصف التوزيع اللوني الكامل للصورة. يمكن إدخاله مباشرةً في أي نموذج تعلم آلة.

٥ تحسينات عملية مهمة

أ. تقليل عدد الخانات (bins)

# بدلاً من 256 خانة، نستخدم 32 خانة
hist = cv2.calcHist([img], [0], None, [32], [0, 256])
# 32 خانة بدلاً من 256 → أصغر وأسرع
# 3 قنوات × 32 = 96 ميزة فقط (بدلاً من 768)

ب. تطبيع الهيستوغرام

# تطبيع ليصبح احتمالاً (مجموعه = 1)
hist = hist / hist.sum()
# هذا يجعله مقاوماً لحجم الصورة
# صورة كبيرة وصورة صغيرة من نفس المشهد ستعطيان نفس الهيستوغرام

ج. استخدام فضاء لون HSV

# فضاء HSV أفضل للهيستوغرام
img_hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
# H = Hue (التدرج اللوني) — القناة الأهم
# S = Saturation (التشبع)
# V = Value (الإضاءة) — أقل أهمية للتعرف
hist_h = cv2.calcHist([img_hsv], [0], None, [180], [0, 180])

البساطة✅ سهل الحساب وسريع
الإضاءة❌ حساس لتغيرات الإضاءة
الموقع❌ لا يأخذ موقع الألوان في الاعتبار
التعرف✅ جيد للتعرف على الأجسام الملونة
الحجم✅ متجه صغير نسبياً (768 أو أقل)

٦ الكود الكامل مع الشرح

import cv2
import numpy as np
import matplotlib.pyplot as plt

# الخطوة 1: قراءة الصورة الملونة
img = cv2.imread('person.jpg')

# الخطوة 2: رسم هيستوغرام القنوات الثلاث
colors = ('b', 'g', 'r')  # ألوان الرسم = ترتيب BGR

plt.figure()
plt.title("Color Histogram")
plt.xlabel("Pixel Intensity")  # المحور الأفقي: قيمة البيكسل (0-255)
plt.ylabel("Count")           # المحور الرأسي: عدد البيكسلات

for i, color in enumerate(colors):
    hist = cv2.calcHist([img], [i], None, [256], [0, 256])
    plt.plot(hist, color=color)

plt.show()

# الخطوة 3: بناء متجه الميزات
hist_b = cv2.calcHist([img], [0], None, [256], [0, 256])
hist_g = cv2.calcHist([img], [1], None, [256], [0, 256])
hist_r = cv2.calcHist([img], [2], None, [256], [0, 256])

feature_vector = np.concatenate([hist_b, hist_g, hist_r])
print(feature_vector.shape)  # (768, 1)
