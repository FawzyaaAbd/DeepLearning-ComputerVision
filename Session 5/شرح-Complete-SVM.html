٣ دوال استخراج الميزات الست

الدالة 1: البيكسلات الخام

def extract_raw_pixels(images):
    features = images.reshape(len(images), -1)
    features = features / 255.0
    return features
reshape(len(images), -1): تحويل (N, 28, 28) إلى (N, 784)
الـ -1 تعني "احسب الحجم تلقائياً" = 28×28 = 784
/ 255.0: تطبيع إلى [0, 1]

الدالة 2: الهيستوغرام الرمادي

def extract_histogram(images, bins=32):
    features = []
    for img in images:
        hist = cv2.calcHist([img.astype(np.uint8)], [0], None, [bins], [0, 256])
        hist = hist.flatten()
        hist = hist / (hist.sum() + 1e-7)  # تطبيع
        features.append(hist)
    return np.array(features)
img.astype(np.uint8): calcHist تتطلب صور uint8، لكن MNIST تأتي كـ uint8 فعلاً
bins=32: 32 خانة بدلاً من 256 لتقليل حجم المتجه
+ 1e-7: نضيف رقماً صغيراً جداً لتجنب القسمة على صفر (Zero Division)

الدالة 3: حواف Canny

def extract_edges(images):
    features = []
    for img in images:
        img_uint8 = img.astype(np.uint8)
        edges = cv2.Canny(img_uint8, 50, 150)
        edge_vector = edges.flatten() / 255.0
        features.append(edge_vector)
    return np.array(features)
العتبتان (50, 150): مناسبتان لصور MNIST الصغيرة (28×28)
/ 255.0: Canny ترجع 0 أو 255، نحولها إلى 0 أو 1

الدالة 4: HOG

def extract_hog(images):
    features = []
    for img in images:
        img_norm = img / 255.0
        hog_feature = hog(img_norm,
                         orientations=9,
                         pixels_per_cell=(7, 7),
                         cells_per_block=(2, 2),
                         block_norm="L2-Hys")
        features.append(hog_feature)
    return np.array(features)
pixels_per_cell=(7, 7): خلية 7×7 لصورة 28×28 → 4×4 = 16 خلية
block_norm="L2-Hys": L2 normalization ثم قص القيم ثم L2 مجدداً — الأكثر استقراراً

الدالة 5: LBP

def extract_lbp(images, radius=1, points=8):
    features = []
    for img in images:
        lbp = local_binary_pattern(img, P=points, R=radius, method="uniform")
        n_bins = points + 2
        hist, _ = np.histogram(lbp.ravel(), bins=n_bins, range=(0, n_bins))
        hist = hist.astype("float")
        hist = hist / (hist.sum() + 1e-7)
        features.append(hist)
    return np.array(features)
np.histogram(lbp.ravel(), bins=n_bins):
— lbp.ravel(): تسطيح مصفوفة LBP إلى صف واحد
— bins=n_bins = P+2 = 10: عدد خانات الهيستوغرام
— range=(0, n_bins): نطاق القيم
— ترجع (hist, bin_edges) — نأخذ hist فقط (_ للحافات)

الدالة 6: HOG + LBP مدموجتان

def extract_hog_lbp(images):
    hog_features = extract_hog(images)
    lbp_features = extract_lbp(images)
    combined = np.hstack([hog_features, lbp_features])
    return combined
np.hstack: دمج أفقي — يلصق متجهات HOG وLBP جنباً إلى جنب.
إذا HOG = (N, 324) وLBP = (N, 10) → HOG+LBP = (N, 334)
الفكرة: HOG قوي في وصف الشكل، وLBP قوي في وصف النسيج — معاً يكملان بعضهما.

٤ حلقة التدريب والتقييم

feature_methods = {
    "Raw Pixels": extract_raw_pixels,
    "Histogram" : extract_histogram,
    "Edges"     : extract_edges,
    "HOG"       : extract_hog,
    "LBP"       : extract_lbp,
    "HOG + LBP" : extract_hog_lbp
}

results = []

for method_name, extractor in feature_methods.items():

    X_train_features = extractor(x_train)
    X_test_features  = extractor(x_test)

    model = Pipeline([
        ("scaler", StandardScaler()),
        ("svm"   , SVC(kernel="linear"))
    ])

    model.fit(X_train_features, y_train)
    y_pred = model.predict(X_test_features)
    acc = accuracy_score(y_test, y_pred)

    results.append({
        "Feature Method"    : method_name,
        "Feature Vector Size": X_train_features.shape[1],
        "Accuracy"          : acc
    })

🔹 feature_methods — قاموس الطرق

قاموس Python (Dictionary) حيث المفتاح هو اسم الطريقة والقيمة هي الدالة نفسها. هذا يسمح بتكرار نفس الكود لكل الطرق بشكل نظيف.

🔹 Pipeline([("scaler", ...), ("svm", ...)])

Pipeline يُسلسل خطوات المعالجة والنموذج:

الخطوة 1 — StandardScaler:
يطبّع الميزات بحيث متوسطها = 0 وانحرافها المعياري = 1.
المعادلة: x' = (x - μ) / σ
SVM حساس جداً لنطاق القيم — بدون تطبيع قد تسود ميزات بقيم كبيرة.

الخطوة 2 — SVC(kernel="linear"):
Support Vector Classifier بنواة خطية.
يجد الحد الفاصل الأمثل بين الفئات بأكبر هامش ممكن.

🔹 feature_methods.items()

تُرجع أزواجاً من (المفتاح، القيمة) — هنا (اسم الطريقة، الدالة). في كل تكرار نستخرج الميزات بالدالة الحالية ونُدرّب SVM عليها.

🔹 X_train_features.shape[1]

حجم متجه الميزات — العدد 1 يعني البُعد الثاني (الأعمدة = عدد الميزات). البُعد الأول (shape[0]) هو عدد العينات.

٦ ما هو SVM؟ — الشرح المعمق

SVM = Support Vector Machine (آلة المتجهات الداعمة)

الفكرة: إيجاد حد فاصل (Hyperplane) يفصل بين الفئات بأكبر هامش ممكن.

المصطلحالشرح
Hyperplaneخط في 2D، مستوى في 3D، فضاء فائق في الأبعاد العليا
Support Vectorsأقرب النقاط لحد الفصل — هي فقط التي تُحدد الحد الفاصل
Marginالمسافة بين الحد الفاصل وأقرب نقاط الفئتين — نريد تعظيمه
kernel="linear"حد فاصل خطي — مناسب عندما الميزات كثيرة والبيانات خطياً قابلة للفصل
kernel="rbf"نواة Radial Basis — تُضاف أبعاداً جديدة لتمكين الفصل غير الخطي
✅ لماذا SVM مناسب مع ميزات HOG؟
HOG تُنتج متجهات عالية الأبعاد (مئات إلى آلاف الميزات). SVM يتفوق في الفضاءات عالية الأبعاد لأنه يعتمد على Support Vectors فقط وليس على كل البيانات.

الدروس المستفادة:

3. دمج الميزات (HOG+LBP) غالباً يُحسّن الأداء
4. الميزات يجب أن تتناسب مع طبيعة المشكلة
